### 3. 代码优化

#### 统一暂停逻辑
```kotlin
// 创建基类ViewModel（可选）
abstract class BaseGameViewModel<T : BaseGameUiState> : ViewModel() {
    protected abstract val _uiState: MutableStateFlow<T>
    protected var timerJob: Job? = null
    
    open fun pauseGame() {
        timerJob?.cancel()
        updatePauseState(true)
    }
    
    open fun resumeGame() {
        updatePauseState(false)
        if (shouldResumeTimer()) {
            startTimer()
        }
    }
    
    open fun restartGame() {
        updatePauseState(false)
        startGame()
    }
    
    protected abstract fun updatePauseState(isPaused: Boolean)
    protected abstract fun shouldResumeTimer(): Boolean
    protected abstract fun startTimer()
    protected abstract fun startGame()
}
```
