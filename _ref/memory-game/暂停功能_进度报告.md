```kotlin
fun pauseGame() {
    timerJob?.cancel()
    gameJob?.cancel()
    _uiState.value = _uiState.value.copy(isPaused = true)
}

fun resumeGame() {
    _uiState.value = _uiState.value.copy(isPaused = false)
    if (_uiState.value.gameState == GameState.USER_INPUT) {
        startTimer()
    }
}

fun restartGame() {
    _uiState.value = _uiState.value.copy(isPaused = false)
    startGame()
}
```
